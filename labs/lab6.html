🐘 Лабораторная работа №6

Проект «Зоопарк» (Методы и интерфейсы)

Время выполнения: 90 минут
Уровень сложности: Продвинутый
Тип: Проект

Цель работы

Изучить методы и интерфейсы в Go, освоить неявную реализацию интерфейсов (duck typing), создать полноценный проект с использованием объектно-ориентированного подхода на Go.

📋 Описание проекта

Вам необходимо создать систему для управления животными в зоопарке. Система должна уметь:

  • Добавлять различных животных
  • Получать информацию о каждом животном
  • Издавать звуки животных
  • Подсчитывать общее количество животных и другие статистические данные

📁 Структура проекта

zoo_project/
├── main.go              # главный файл с функцией main
├── animals/             # пакет с животными
│   ├── animal.go        # интерфейс Animal
│   ├── dog.go           # структура Dog
│   ├── cat.go           # структура Cat
│   ├── bird.go          # структура Bird
│   └── zoo.go           # структура Zoo для управления животными
└── go.mod               # модуль проекта (создаётся автоматически)

🔨 Этап 1. Создание интерфейса Animal

1 Создайте файл animals/animal.go с интерфейсом Animal, который содержит следующие методы:

type Animal interface {
    Speak() string        // звук, который издаёт животное
    Move() string         // как животное двигается
    GetInfo() string      // полная информация о животном
    GetName() string      // имя животного
    GetType() string      // тип животного (собака, кошка и т.д.)
}

Файл animals/bird.go

package animals

import "fmt"

type Bird struct {
    Name   string
    CanFly bool
    Age    int
}

func (b Bird) Speak() string {
    if b.CanFly {
        return "Чирик-чирик!"
    }
    return "Ку-ка-ре-ку!"
}

func (b Bird) Move() string {
    if b.CanFly {
        return "летает по небу"
    }
    return "ходит по земле"
}

func (b Bird) GetName() string {
    return b.Name
}

func (b Bird) GetType() string {
    if b.CanFly {
        return "Птица (летающая)"
    }
    return "Птица (нелетающая)"
}

func (b Bird) GetInfo() string {
    flyStatus := "летает"
    if !b.CanFly {
        flyStatus = "не летает"
    }
    return fmt.Sprintf("%s (%s), возраст: %d лет, %s", 
        b.GetType(), b.Name, b.Age, flyStatus)
}

🔨 Этап 3. Создание структуры Zoo

3 Создайте файл animals/zoo.go со структурой Zoo и методами управления животными:

package animals

import "fmt"

type Zoo struct {
    Name    string
    animals []Animal
}

func NewZoo(name string) *Zoo {
    return &Zoo{
        Name:    name,
        animals: make([]Animal, 0),
    }
}

func (z *Zoo) AddAnimal(a Animal) {
    z.animals = append(z.animals, a)
    fmt.Printf("Животное '%s' добавлено в зоопарк '%s'\n", a.GetName(), z.Name)
}

func (z *Zoo) RemoveAnimal(name string) bool {
    for i, a := range z.animals {
        if a.GetName() == name {
            z.animals = append(z.animals[:i], z.animals[i+1:]...)
            fmt.Printf("Животное '%s' удалено из зоопарка\n", name)
            return true
        }
    }
    fmt.Printf("Животное '%s' не найдено\n", name)
    return false
}

func (z *Zoo) FindAnimal(name string) Animal {
    for _, a := range z.animals {
        if a.GetName() == name {
            return a
        }
    }
    return nil
}

func (z *Zoo) ListAllAnimals() {
    if len(z.animals) == 0 {
        fmt.Println("В зоопарке нет животных")
        return
    }
    
    fmt.Printf("\n=== Животные в зоопарке '%s' ===\n", z.Name)
    for i, a := range z.animals {
        fmt.Printf("%d. %s\n", i+1, a.GetInfo())
        fmt.Printf("   Звук: %s\n", a.Speak())
        fmt.Printf("   Движение: %s\n", a.Move())
        fmt.Println()
    }
}

func (z *Zoo) GetTotalAnimals() int {
    return len(z.animals)
}

func (z *Zoo) MakeAllSounds() {
    fmt.Printf("\n=== Все животные зоопарка '%s' издают звуки ===\n", z.Name)
    for _, a := range z.animals {
        fmt.Printf("%s: %s\n", a.GetName(), a.Speak())
    }
}

🔨 Этап 4. Главная программа

4 Создайте файл main.go:

package main

import (
    "bufio"
    "fmt"
    "os"
    "strconv"
    "strings"
    "zoo_project/animals"
)

func main() {
    // Создание зоопарка
    zoo := animals.NewZoo("Московский зоопарк")
    
    // Добавление животных
    fmt.Println("=== Добавление животных в зоопарк ===\n")
    
    dog := animals.Dog{Name: "Шарик", Breed: "Овчарка", Age: 5}
    cat := animals.Cat{Name: "Мурка", Color: "Рыжая", Age: 3}
    bird1 := animals.Bird{Name: "Кеша", CanFly: true, Age: 2}
    bird2 := animals.Bird{Name: "Петя", CanFly: false, Age: 1}
    
    zoo.AddAnimal(dog)
    zoo.AddAnimal(cat)
    zoo.AddAnimal(bird1)
    zoo.AddAnimal(bird2)
    
    // Вывод всех животных
    zoo.ListAllAnimals()
    
    // Все животные издают звуки
    zoo.MakeAllSounds()
    
    // Поиск животного
    fmt.Println("\n=== Поиск животного ===")
    reader := bufio.NewReader(os.Stdin)
    fmt.Print("Введите имя животного для поиска: ")
    name, _ := reader.ReadString('\n')
    name = strings.TrimSpace(name)
    
    found := zoo.FindAnimal(name)
    if found != nil {
        fmt.Printf("Найдено: %s\n", found.GetInfo())
    } else {
        fmt.Printf("Животное '%s' не найдено\n", name)
    }
    
    // Удаление животного
    fmt.Println("\n=== Удаление животного ===")
    fmt.Print("Введите имя животного для удаления: ")
    removeName, _ := reader.ReadString('\n')
    removeName = strings.TrimSpace(removeName)
    zoo.RemoveAnimal(removeName)
    
    // Статистика
    fmt.Printf("\n=== Статистика зоопарка ===\n")
    fmt.Printf("Всего животных: %d\n", zoo.GetTotalAnimals())
}

⭐ Дополнительные задания

+1 Статистика по зоопарку

Добавьте в структуру Zoo следующие методы:

  • GetOldestAnimal() Animal — самое старое животное
  • GetYoungestAnimal() Animal — самое молодое животное
  • GetAverageAge() float64 — средний возраст
  • GroupByType() map[string][]Animal — группировка по типу

+2 Сохранение в файл

Добавьте методы для сохранения и загрузки данных зоопарка в файл:

  • SaveToFile(filename string) error
  • LoadZooFromFile(filename string) (*Zoo, error)

Используйте encoding/json для сериализации данных.

+3 Новые типы животных

Добавьте в проект один или несколько новых типов животных: Fish, Snake, Monkey, Elephant или Penguin. Реализуйте для них все методы интерфейса Animal.

Итог лабораторной работы

После выполнения лабораторной работы студент понимает концепцию методов в Go, умеет создавать методы как для структур, так и для любых пользовательских типов. Студент понимает разницу между получателем-значением и получателем-указателем и умеет выбирать правильный тип получателя в зависимости от задачи. Студент освоил интерфейсы как наборы методов, понимает ключевую особенность Go — неявную реализацию интерфейсов (duck typing), благодаря которой тип автоматически реализует интерфейс, если имеет все его методы. В рамках проекта «Зоопарк» студент создал иерархию животных с общим интерфейсом, реализовал полиморфное поведение, научился управлять коллекцией объектов через интерфейсный тип. Студент умеет добавлять новые типы животных без изменения существующего кода, что демонстрирует преимущества использования интерфейсов для создания расширяемых систем.
