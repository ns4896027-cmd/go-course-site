🌐 Лабораторная работа №7

Проект «Конкурентный веб-сервис» (URL Health Checker)

Время выполнения: 90 минут
Уровень сложности: Продвинутый
Тип: Проект

Цель работы

Изучить конкурентное программирование в Go и создать веб-сервис с использованием горутин, каналов и контекста для параллельной обработки HTTP-запросов.

📋 Описание проекта

Вам необходимо создать веб-сервис — «Анализатор веб-страниц» (URL Health Checker), который:

  • Принимает список URL-адресов через HTTP-запрос
  • Параллельно проверяет доступность каждого URL (делает HTTP-запрос)
  • Возвращает результаты проверки (статус, время ответа, код ответа)
  • Обрабатывает ошибки и таймауты
  • Предоставляет эндпоинт для получения статистики

🔨 Этап 1. Создание структур данных

1 Создайте файл models/models.go со структурами данных:

package models

import "time"

// Запрос на проверку URL
type CheckRequest struct {
    URLs []string `json:"urls"`
}

// Результат проверки одного URL
type URLResult struct {
    URL        string        `json:"url"`
    Status     string        `json:"status"`      // "OK", "ERROR", "TIMEOUT"
    StatusCode int           `json:"status_code"`
    Latency    time.Duration `json:"latency_ms"`
    Error      string        `json:"error,omitempty"`
    CheckedAt  time.Time     `json:"checked_at"`
}

// Ответ от сервера
type CheckResponse struct {
    RequestID  string       `json:"request_id"`
    TotalURLs  int          `json:"total_urls"`
    CheckedAt  time.Time    `json:"checked_at"`
    Results    []URLResult  `json:"results"`
    Statistics Statistics   `json:"statistics"`
}

// Статистика проверки
type Statistics struct {
    TotalChecked     int     `json:"total_checked"`
    SuccessCount     int     `json:"success_count"`      // 2xx
    ClientErrorCount int     `json:"client_error_count"` // 4xx
    ServerErrorCount int     `json:"server_error_count"` // 5xx
    TimeoutCount     int     `json:"timeout_count"`
    AvgLatencyMs     float64 `json:"avg_latency_ms"`
}

🔨 Этап 2. Логика проверки URL

2 Создайте файл checker/url_checker.go с логикой параллельной проверки URL:

package checker

import (
    "context"
    "fmt"
    "net/http"
    "sync"
    "time"
    "url_checker/models"
)

type CheckerConfig struct {
    Timeout       time.Duration
    MaxConcurrent int
    MaxURLs       int
}

type URLChecker struct {
    client *http.Client
    config CheckerConfig
}

func NewURLChecker(config CheckerConfig) *URLChecker {
    return &URLChecker{
        client: &http.Client{Timeout: config.Timeout},
        config: config,
    }
}

func (uc *URLChecker) checkSingleURL(ctx context.Context, url string) models.URLResult {
    result := models.URLResult{
        URL:       url,
        CheckedAt: time.Now(),
    }
    start := time.Now()

    req, err := http.NewRequestWithContext(ctx, "GET", url, nil)
    if err != nil {
        result.Status = "ERROR"
        result.Error = fmt.Sprintf("Ошибка запроса: %v", err)
        result.Latency = time.Since(start)
        return result
    }

    resp, err := uc.client.Do(req)
    result.Latency = time.Since(start)

    if err != nil {
        result.Status = "ERROR"
        result.Error = err.Error()
        return result
    }
    defer resp.Body.Close()

    result.StatusCode = resp.StatusCode

    switch {
    case resp.StatusCode >= 200 && resp.StatusCode < 300:
        result.Status = "OK"
    case resp.StatusCode >= 400 && resp.StatusCode < 500:
        result.Status = "CLIENT_ERROR"
    case resp.StatusCode >= 500:
        result.Status = "SERVER_ERROR"
    default:
        result.Status = "UNKNOWN"
    }
    return result
}

func (uc *URLChecker) CheckURLs(ctx context.Context, urls []string) ([]models.URLResult, models.Statistics) {
    if len(urls) > uc.config.MaxURLs {
        urls = urls[:uc.config.MaxURLs]
    }

    results := make([]models.URLResult, len(urls))
    stats := models.Statistics{}

    semaphore := make(chan struct{}, uc.config.MaxConcurrent)
    var wg sync.WaitGroup
    var mu sync.Mutex

    for i, url := range urls {
        wg.Add(1)
        go func(idx int, u string) {
            defer wg.Done()
            select {
            case semaphore <- struct{}{}:
                defer func() { <-semaphore }()
            case <-ctx.Done():
                return
            }
            result := uc.checkSingleURL(ctx, u)
            results[idx] = result

            mu.Lock()
            stats.TotalChecked++
            switch result.Status {
            case "OK":
                stats.SuccessCount++
            case "CLIENT_ERROR":
                stats.ClientErrorCount++
            case "SERVER_ERROR":
                stats.ServerErrorCount++
            case "ERROR":
                stats.TimeoutCount++
            }
            stats.AvgLatencyMs = (stats.AvgLatencyMs*float64(stats.TotalChecked-1) +
                float64(result.Latency.Milliseconds())) / float64(stats.TotalChecked)
            mu.Unlock()
        }(i, url)
    }
    wg.Wait()
    return results, stats
}

🔨 Этап 3. HTTP-обработчики

3 Создайте файл handlers/check.go с обработчиками HTTP-запросов:

package handlers

import (
    "encoding/json"
    "fmt"
    "net/http"
    "time"
    "url_checker/checker"
    "url_checker/models"
)

type CheckHandler struct {
    urlChecker *checker.URLChecker
}

func NewCheckHandler(c *checker.URLChecker) *CheckHandler {
    return &CheckHandler{urlChecker: c}
}

func (h *CheckHandler) CheckURLs(w http.ResponseWriter, r *http.Request) {
    if r.Method != http.MethodPost {
        http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
        return
    }

    var req models.CheckRequest
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        http.Error(w, "Invalid JSON", http.StatusBadRequest)
        return
    }

    if len(req.URLs) == 0 {
        http.Error(w, "No URLs provided", http.StatusBadRequest)
        return
    }

    results, stats := h.urlChecker.CheckURLs(r.Context(), req.URLs)

    response := models.CheckResponse{
        RequestID:  fmt.Sprintf("%d", time.Now().UnixNano()),
        TotalURLs:  len(req.URLs),
        CheckedAt:  time.Now(),
        Results:    results,
        Statistics: stats,
    }

    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(response)
}

func HealthCheck(w http.ResponseWriter, r *http.Request) {
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(map[string]string{
        "status": "ok",
        "time":   time.Now().Format(time.RFC3339),
    })
}

🔨 Этап 4. Главная программа

4 Создайте файл main.go:

package main

import (
    "context"
    "log"
    "net/http"
    "os"
    "os/signal"
    "time"
    "url_checker/checker"
    "url_checker/handlers"
)

func main() {
    config := checker.CheckerConfig{
        Timeout:       10 * time.Second,
        MaxConcurrent: 5,
        MaxURLs:       20,
    }

    urlChecker := checker.NewURLChecker(config)
    checkHandler := handlers.NewCheckHandler(urlChecker)

    mux := http.NewServeMux()
    mux.HandleFunc("/check", checkHandler.CheckURLs)
    mux.HandleFunc("/health", handlers.HealthCheck)

    server := &http.Server{
        Addr:    ":8080",
        Handler: mux,
    }

    go func() {
        log.Printf("Сервер запущен на http://localhost:8080")
        log.Println("Доступные эндпоинты:")
        log.Println("  POST /check - проверить список URL")
        log.Println("  GET /health - проверка здоровья")
        if err := server.ListenAndServe(); err != nil && err != http.ErrServerClosed {
            log.Fatalf("Ошибка: %v", err)
        }
    }()

    // Graceful shutdown
    quit := make(chan os.Signal, 1)
    signal.Notify(quit, os.Interrupt)
    <-quit

    log.Println("Завершение работы...")
    ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
    defer cancel()
    if err := server.Shutdown(ctx); err != nil {
        log.Fatalf("Ошибка остановки: %v", err)
    }
    log.Println("Сервер остановлен")
}

⭐ Дополнительные задания

+1 Эндпоинт статистики

Добавьте в структуру URLChecker накопление статистики по всем проверкам и создайте эндпоинт /stats для получения накопленной статистики.

+2 Таймаут для всего запроса

Добавьте возможность ограничения времени выполнения всего запроса через заголовок X-Request-Timeout (в секундах).

+3 Отмена долгих операций

Реализуйте возможность отмены долгой операции через эндпоинт /cancel/{request_id}. Для этого нужно генерировать уникальный ID для каждого запроса и сохранять контекст.

+4 CLI-клиент

Создайте отдельную программу (CLI-клиент), которая принимает список URL через аргументы командной строки, отправляет запрос к сервису и красиво выводит результаты.

Итог лабораторной работы

После выполнения лабораторной работы студент понимает разницу между конкурентностью и параллелизмом, умеет запускать горутины с помощью ключевого слова go. Студент освоил каналы как основной механизм общения между горутинами, знает разницу между буферизированными и небуферизированными каналами, умеет использовать оператор select для ожидания нескольких каналов одновременно. Студент владеет приёмами синхронизации горутин с помощью WaitGroup и мьютексов, понимает проблему состояния гонки и умеет ограничивать количество параллельно выполняемых операций с помощью семафоров. Студент изучил пакет context для управления отменой операций и установки таймаутов. В рамках проекта студент создал полноценный REST API на стандартной библиотеке net/http, реализовал параллельную проверку доступности URL с заданными ограничениями, организовал graceful shutdown сервера для корректного завершения работы. Студент умеет работать с JSON (маршалинг и анмаршалинг), создавать middleware-обработчики, структурировать проект на пакеты. В результате студент готов разрабатывать конкурентные веб-сервисы на Go с эффективным использованием ресурсов и высокой производительностью.
