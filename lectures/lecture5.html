Лекция 5. Карты (maps) и структуры

Длительность лекции: 60 минут
Уровень: Средний
Темы: Карты (maps), операции с картами, итерация, структуры (structs)

🗺️ Что такое карта (map)?

Карта (map) — это ссылка на хеш-таблицу — структуру данных, где каждый элемент представляет пару «ключ-значение». Каждый элемент имеет уникальный ключ, по которому можно получить значение.

Аналогия: Карта — это как телефонная книга. По имени (ключу) вы находите номер телефона (значение).

📦 Создание карт

Объявление (nil-карта)

var m map[string]int
fmt.Println(m == nil)  // true
// m["key"] = 42 // panic! нельзя добавлять в nil-карту

Создание с помощью make()

// Пустая карта (готова к использованию)
scores := make(map[string]int)

// Можно задать начальную емкость (оптимизация)
phones := make(map[string]string, 100)

scores["Анна"] = 95
scores["Борис"] = 87

Литерал карты (инициализация при создании)

grades := map[string]int{
    "Математика": 5,
    "Физика":     4,
    "Информатика": 5,
}

// Пустая карта через литерал
empty := map[string]int{}

Сравнение способов создания

СпособКодМожно добавлятьКогда использовать
Объявлениеvar m map[K]V❌ НетТолько для чтения
makem := make(map[K]V)✅ ДаДинамическое заполнение
Литералm := map[K]V{}✅ ДаИзвестные начальные данные

⚙️ Основные операции с картами

Добавление и изменение элементов

students := make(map[int]string)

// Добавление
students[101] = "Анна Петрова"
students[102] = "Борис Сидоров"

// Изменение (тот же синтаксис)
students[101] = "Анна Смирнова"  // перезапись

Чтение элементов

// Простое чтение (если ключа нет — zero value)
name := students[101]
unknown := students[999]  // "" (пустая строка)

// Проверка существования ключа (comma ok idiom)
value, exists := students[102]
if exists {
    fmt.Println("Найден:", value)
} else {
    fmt.Println("Не найден")
}

Удаление элементов (delete)

students := map[int]string{
    101: "Анна",
    102: "Борис",
    103: "Виктор",
}

delete(students, 102)  // удаляем Бориса
fmt.Println(students)   // map[101:Анна 103:Виктор]

// Удаление несуществующего ключа — безопасно
delete(students, 999)  // нет паники

Длина и очистка карты

phones := map[string]string{"a": "123", "b": "456"}
fmt.Println(len(phones))  // 2

// Очистка (Go 1.21+)
clear(phones)

// Старый способ (до Go 1.21)
for k := range phones {
    delete(phones, k)
}

🔄 Перебор карты (range)

Базовый перебор

capitals := map[string]string{
    "Россия":   "Москва",
    "Франция":  "Париж",
    "Германия": "Берлин",
}

for country, capital := range capitals {
    fmt.Printf("%s → %s\n", country, capital)
}
Важно! Порядок перебора карты не гарантируется и может быть разным при каждом запуске!

Упорядоченный вывод

// Собираем ключи и сортируем
countries := make([]string, 0, len(capitals))
for country := range capitals {
    countries = append(countries, country)
}

sort.Strings(countries)

for _, country := range countries {
    fmt.Printf("%s → %s\n", country, capitals[country])
}

Перебор только ключей или только значений

// Только ключи
for country := range capitals {
    fmt.Println(country)
}

// Только значения
for _, capital := range capitals {
    fmt.Println(capital)
}

🏗️ Структуры (struct)

Структура — это составной тип данных, который объединяет несколько полей (переменных) в одну сущность.

Определение структуры

type Person struct {
    Name string
    Age  int
    City string
}

func main() {
    // Создание экземпляра структуры
    person1 := Person{
        Name: "Анна",
        Age:  25,
        City: "Москва",
    }
    
    person2 := Person{"Борис", 30, "СПб"}
    
    // Доступ к полям
    fmt.Println(person1.Name)  // Анна
    fmt.Println(person2.Age)   // 30
    
    // Изменение полей
    person1.City = "Санкт-Петербург"
}

Вложенные структуры

type Address struct {
    City    string
    Street  string
    ZipCode string
}

type Student struct {
    Name    string
    Age     int
    Address Address
}

func main() {
    student := Student{
        Name: "Анна",
        Age:  20,
        Address: Address{
            City:    "Москва",
            Street:  "Тверская",
            ZipCode: "101000",
        },
    }
    
    fmt.Println(student.Address.City)  // Москва
}

💡 Практические примеры

Пример 1. Счётчик слов

func wordCount(text string) map[string]int {
    words := strings.Fields(text)
    count := make(map[string]int)
    
    for _, word := range words {
        word = strings.ToLower(strings.Trim(word, ".,!?;:"))
        count[word]++
    }
    return count
}

func main() {
    text := "To be or not to be, that is the question. To be!"
    stats := wordCount(text)
    
    for word, cnt := range stats {
        fmt.Printf("%s: %d\n", word, cnt)
    }
}

Пример 2. Группировка данных по структурам

type Person struct {
    Name string
    Age  int
    City string
}

func groupByCity(people []Person) map[string][]Person {
    groups := make(map[string][]Person)
    
    for _, p := range people {
        groups[p.City] = append(groups[p.City], p)
    }
    return groups
}

func main() {
    people := []Person{
        {"Анна", 25, "Москва"},
        {"Борис", 30, "СПб"},
        {"Виктор", 28, "Москва"},
        {"Галина", 35, "СПб"},
        {"Дмитрий", 22, "Казань"},
    }
    
    byCity := groupByCity(people)
    
    for city, persons := range byCity {
        fmt.Printf("Город %s:\n", city)
        for _, p := range persons {
            fmt.Printf("  - %s (%d лет)\n", p.Name, p.Age)
        }
    }
}
