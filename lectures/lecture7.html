Лекция 7. Обработка ошибок и работа с ресурсами

Длительность лекции: 50 минут
Уровень: Продвинутый
Темы: Подход Go к ошибкам, создание ошибок, оборачивание ошибок, работа с файлами, пакеты io/os, пакет time

🎯 Подход Go к обработке ошибок

В отличие от многих языков (Java, Python, C++), в Go нет исключений (exceptions). Вместо этого используется явный возврат ошибок как обычных значений.

Философия Go: Ошибки — это значения. Обрабатывайте их так же, как и любые другие данные.

Базовый паттерн обработки ошибок

result, err := someFunction()
if err != nil {
    // Обрабатываем ошибку
    log.Fatal(err)
}
// Используем result
Важное правило: Всегда проверяйте ошибки! Никогда не игнорируйте их (например, через _) без веской причины.

Стандартный тип error

type error interface {
    Error() string
}

Любой тип, реализующий метод Error() string, автоматически удовлетворяет интерфейсу error.

📝 Создание ошибок

errors.New() — простая ошибка

import "errors"

func divide(a, b float64) (float64, error) {
    if b == 0 {
        return 0, errors.New("деление на ноль")
    }
    return a / b, nil
}

func main() {
    result, err := divide(10, 0)
    if err != nil {
        fmt.Println("Ошибка:", err)  // Ошибка: деление на ноль
        return
    }
    fmt.Println("Результат:", result)
}

fmt.Errorf() — форматированная ошибка

func readFile(filename string) error {
    if filename == "" {
        return fmt.Errorf("имя файла не может быть пустым")
    }
    return nil
}

Пользовательские типы ошибок

type ValidationError struct {
    Field   string
    Value   interface{}
    Message string
}

func (e ValidationError) Error() string {
    return fmt.Sprintf("ошибка валидации поля '%s' (значение %v): %s",
        e.Field, e.Value, e.Message)
}

func validateAge(age int) error {
    if age < 0 {
        return ValidationError{
            Field:   "age",
            Value:   age,
            Message: "возраст не может быть отрицательным",
        }
    }
    if age > 150 {
        return ValidationError{
            Field:   "age",
            Value:   age,
            Message: "возраст не может превышать 150 лет",
        }
    }
    return nil
}

📦 Оборачивание ошибок (error wrapping)

Оборачивание позволяет добавлять контекст к ошибке, сохраняя исходную информацию.

Оборачивание с %w

func processFile(filename string) error {
    data, err := os.ReadFile(filename)
    if err != nil {
        return fmt.Errorf("не удалось прочитать файл %s: %w", filename, err)
    }
    // обработка данных
    return nil
}

Проверка обёрнутой ошибки (errors.Is)

import "errors"

var ErrNotFound = errors.New("not found")

func findUser(id int) error {
    return fmt.Errorf("ошибка поиска пользователя %d: %w", id, ErrNotFound)
}

func main() {
    err := findUser(42)
    if errors.Is(err, ErrNotFound) {
        fmt.Println("Пользователь не найден")
    }
}

Извлечение типа ошибки (errors.As)

func handleError(err error) {
    var valErr ValidationError
    if errors.As(err, &valErr) {
        fmt.Printf("Ошибка валидации поля %s: %s\n", valErr.Field, valErr.Message)
    } else {
        fmt.Println("Другая ошибка:", err)
    }
}

🔧 Defer — отложенные вызовы

defer откладывает выполнение функции до момента возврата из объемлющей функции. Это удобно для освобождения ресурсов.

func readFile(filename string) (string, error) {
    file, err := os.Open(filename)
    if err != nil {
        return "", err
    }
    defer file.Close()  // файл закроется при выходе из функции
    
    data, err := io.ReadAll(file)
    if err != nil {
        return "", err
    }
    
    return string(data), nil
}
Важно: defer работает по принципу LIFO (последний зарегистрированный — первый выполняется). 
func main() {
    defer fmt.Println("Первый")
    defer fmt.Println("Второй")
    defer fmt.Println("Третий")
    fmt.Println("Начало")
}
// Вывод:
// Начало
// Третий
// Второй
// Первый

📄 Работа с файлами (пакеты os и io)

Чтение файла

import (
    "os"
    "io"
)

func readFileExample(filename string) error {
    file, err := os.Open(filename)
    if err != nil {
        return fmt.Errorf("ошибка открытия файла: %w", err)
    }
    defer file.Close()
    
    // Чтение всего файла
    data, err := io.ReadAll(file)
    if err != nil {
        return fmt.Errorf("ошибка чтения: %w", err)
    }
    
    fmt.Println(string(data))
    return nil
}

Запись в файл

func writeFileExample(filename, content string) error {
    file, err := os.Create(filename)
    if err != nil {
        return fmt.Errorf("ошибка создания файла: %w", err)
    }
    defer file.Close()
    
    _, err = file.WriteString(content)
    if err != nil {
        return fmt.Errorf("ошибка записи: %w", err)
    }
    
    return nil
}

Буферизированное чтение (bufio)

import "bufio"

func readLines(filename string) ([]string, error) {
    file, err := os.Open(filename)
    if err != nil {
        return nil, err
    }
    defer file.Close()
    
    var lines []string
    scanner := bufio.NewScanner(file)
    for scanner.Scan() {
        lines = append(lines, scanner.Text())
    }
    
    if err := scanner.Err(); err != nil {
        return nil, err
    }
    
    return lines, nil
}

Проверка существования файла

func fileExists(filename string) bool {
    _, err := os.Stat(filename)
    return !os.IsNotExist(err)
}

⏰ Работа с временем (пакет time)

Текущее время

now := time.Now()
fmt.Println(now)                       // 2024-01-15 14:30:00 +0300 MSK
fmt.Println(now.Year())                // 2024
fmt.Println(now.Month())               // January
fmt.Println(now.Day())                 // 15
fmt.Println(now.Hour())                // 14
fmt.Println(now.Minute())              // 30

Форматирование времени

// Go использует "магическую" дату: Mon Jan 2 15:04:05 MST 2006
fmt.Println(now.Format("2006-01-02 15:04:05"))           // 2024-01-15 14:30:00
fmt.Println(now.Format("02.01.2006"))                    // 15.01.2024
fmt.Println(now.Format("15:04"))                         // 14:30
fmt.Println(now.Format("Monday, January 2"))             // Monday, January 15

Парсинг времени из строки

layout := "2006-01-02 15:04:05"
str := "2024-12-25 10:30:00"
parsedTime, err := time.Parse(layout, str)
if err != nil {
    fmt.Println("Ошибка парсинга:", err)
    return
}
fmt.Println(parsedTime)  // 2024-12-25 10:30:00 +0000 UTC

Таймеры и задержки

// Задержка (time.Sleep)
time.Sleep(2 * time.Second)
fmt.Println("Прошло 2 секунды")

// Таймер
timer := time.NewTimer(3 * time.Second)
<-timer.C
fmt.Println("Таймер сработал")

// Таймер с остановкой
timer2 := time.NewTimer(5 * time.Second)
time.Sleep(2 * time.Second)
timer2.Stop()
fmt.Println("Таймер остановлен")

// Ticker (периодическое выполнение)
ticker := time.NewTicker(1 * time.Second)
go func() {
    for range ticker.C {
        fmt.Println("Тик")
    }
}()
time.Sleep(5 * time.Second)
ticker.Stop()

Измерение времени выполнения

func measureTime() {
    start := time.Now()
    
    // Какой-то код
    time.Sleep(100 * time.Millisecond)
    
    elapsed := time.Since(start)
    fmt.Printf("Выполнение заняло %s\n", elapsed)
    fmt.Printf("Миллисекунды: %d\n", elapsed.Milliseconds())
}

💡 Практический пример: логгер с временными метками

type Logger struct {
    file *os.File
}

func NewLogger(filename string) (*Logger, error) {
    file, err := os.OpenFile(filename, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0644)
    if err != nil {
        return nil, fmt.Errorf("не удалось открыть лог-файл: %w", err)
    }
    
    return &Logger{file: file}, nil
}

func (l *Logger) Log(message string) error {
    timestamp := time.Now().Format("2006-01-02 15:04:05")
    line := fmt.Sprintf("[%s] %s\n", timestamp, message)
    
    _, err := l.file.WriteString(line)
    if err != nil {
        return fmt.Errorf("ошибка записи в лог: %w", err)
    }
    return nil
}

func (l *Logger) Close() error {
    return l.file.Close()
}

func main() {
    logger, err := NewLogger("app.log")
    if err != nil {
        log.Fatal(err)
    }
    defer logger.Close()
    
    logger.Log("Приложение запущено")
    logger.Log("Пользователь авторизован")
    time.Sleep(2 * time.Second)
    logger.Log("Приложение завершено")
}
