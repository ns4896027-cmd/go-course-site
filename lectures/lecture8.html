🔄 Конкурентность vs Параллелизм

Роб Пайк (создатель Go): "Конкурентность — это структура программы. Параллелизм — это выполнение."
ХарактеристикаКонкурентностьПараллелизм
ОпределениеСпособность выполнять несколько задач, переключаясь между нимиОдновременное выполнение на разных ядрах
Количество ядерНе требует многоядерностиТребует многоядерного процессора
АналогияОдин человек, который готовит и убирает (переключается)Несколько человек — каждый занимается своим делом

🚀 Горутины (goroutines)

Горутина — легковесный поток выполнения, управляемый рантаймом Go. В отличие от потоков ОС (1-2 МБ), горутина занимает около 2 КБ памяти.

Запуск горутины

package main

import (
    "fmt"
    "time"
)

func sayHello() {
    fmt.Println("Привет из горутины!")
}

func main() {
    // Запуск горутины
    go sayHello()
    
    // Запуск анонимной функции в горутине
    go func() {
        fmt.Println("Анонимная горутина")
    }()
    
    // Даём время горутинам выполниться
    time.Sleep(100 * time.Millisecond)
    fmt.Println("Главная функция завершена")
}
Важно! Если главная функция (main) завершится, все горутины будут принудительно остановлены.

Пример многозадачности

func printNumbers() {
    for i := 1; i <= 5; i++ {
        fmt.Printf("%d ", i)
        time.Sleep(100 * time.Millisecond)
    }
}

func printLetters() {
    for i := 'A'; i <= 'E'; i++ {
        fmt.Printf("%c ", i)
        time.Sleep(150 * time.Millisecond)
    }
}

func main() {
    go printNumbers()
    go printLetters()
    
    time.Sleep(1 * time.Second)
    fmt.Println("\nГотово!")
}
// Вывод (порядок может меняться): 1 A 2 3 B 4 C 5 D E

📡 Каналы (channels)

Канал — механизм для безопасной передачи данных между горутинами.

Девиз Go: "Не общайтесь через разделяемую память; делитесь памятью через общение"

Создание каналов

ch := make(chan int)          // небуферизированный
ch := make(chan int, 10)      // буферизированный (ёмкость 10)

Небуферизированные каналы

Отправка блокирует отправителя до получения. Получение блокирует получателя до отправки.

func main() {
    ch := make(chan int)
    
    go func() {
        ch <- 42  // блокируется, пока не прочитают
        fmt.Println("Значение отправлено")
    }()
    
    value := <-ch  // блокируется, пока не отправят
    fmt.Println("Получено:", value)
}

Буферизированные каналы

Имеют очередь. Отправка блокируется только когда буфер полон.

func main() {
    ch := make(chan string, 3)
    
    ch <- "первый"
    ch <- "второй"
    ch <- "третий"
    // ch <- "четвёртый"  // заблокируется — буфер полон
    
    fmt.Println(<-ch)  // первый
    fmt.Println(<-ch)  // второй
    fmt.Println(<-ch)  // третий
}

Закрытие каналов и range

func producer(ch chan int) {
    for i := 1; i <= 5; i++ {
        ch <- i
    }
    close(ch)  // закрываем канал
}

func main() {
    ch := make(chan int)
    go producer(ch)
    
    for value := range ch {  // range завершается при закрытии
        fmt.Println("Получено:", value)
    }
    
    // Проверка, закрыт ли канал
    value, ok := <-ch
    fmt.Printf("value=%d, ok=%t\n", value, ok)  // value=0, ok=false
}

🎯 Оператор select

select позволяет ожидать несколько каналов одновременно. Выполняется первый готовый канал.

func main() {
    ch1 := make(chan string)
    ch2 := make(chan string)
    
    go func() {
        time.Sleep(1 * time.Second)
        ch1 <- "из канала 1"
    }()
    
    go func() {
        time.Sleep(2 * time.Second)
        ch2 <- "из канала 2"
    }()
    
    for i := 0; i < 2; i++ {
        select {
        case msg1 := <-ch1:
            fmt.Println("Получено:", msg1)
        case msg2 := <-ch2:
            fmt.Println("Получено:", msg2)
        }
    }
}
// Вывод:
// Получено: из канала 1
// Получено: из канала 2

select с таймаутом

select {
case result := <-ch:
    fmt.Println("Результат:", result)
case <-time.After(3 * time.Second):
    fmt.Println("Таймаут! Операция заняла слишком много времени")
}

select с default (неблокирующая операция)

select {
case msg := <-ch:
    fmt.Println("Получено:", msg)
default:
    fmt.Println("Нет данных в канале")
}

⏱️ Пакет context

context — механизм для отмены операций, установки таймаутов и передачи значений.

Создание контекста

// Пустой контекст
ctx := context.Background()
ctx := context.TODO()

// Контекст с отменой
ctx, cancel := context.WithCancel(context.Background())
defer cancel()

// Контекст с таймаутом
ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
defer cancel()

// Контекст с дедлайном
ctx, cancel := context.WithDeadline(context.Background(), time.Now().Add(5*time.Second))
defer cancel()

Отмена операций

func worker(ctx context.Context, id int) {
    for {
        select {
        case <-ctx.Done():
            fmt.Printf("Воркер %d: остановлен (%v)\n", id, ctx.Err())
            return
        default:
            fmt.Printf("Воркер %d: работает...\n", id)
            time.Sleep(500 * time.Millisecond)
        }
    }
}

func main() {
    ctx, cancel := context.WithCancel(context.Background())
    
    go worker(ctx, 1)
    go worker(ctx, 2)
    
    time.Sleep(2 * time.Second)
    fmt.Println("Отменяем все операции...")
    cancel()
    
    time.Sleep(500 * time.Millisecond)
}

🌐 Создание веб-сервисов (пакет net/http)

Простейший HTTP-сервер

package main

import (
    "fmt"
    "net/http"
)

func helloHandler(w http.ResponseWriter, r *http.Request) {
    fmt.Fprintf(w, "Hello, World!")
}

func aboutHandler(w http.ResponseWriter, r *http.Request) {
    fmt.Fprintf(w, "Это страница About")
}

func main() {
    http.HandleFunc("/", helloHandler)
    http.HandleFunc("/about", aboutHandler)
    
    fmt.Println("Сервер запущен на http://localhost:8080")
    http.ListenAndServe(":8080", nil)
}

Разные HTTP-методы

func userHandler(w http.ResponseWriter, r *http.Request) {
    switch r.Method {
    case http.MethodGet:
        fmt.Fprintf(w, "GET: Получение пользователя")
    case http.MethodPost:
        fmt.Fprintf(w, "POST: Создание пользователя")
    case http.MethodPut:
        fmt.Fprintf(w, "PUT: Обновление пользователя")
    case http.MethodDelete:
        fmt.Fprintf(w, "DELETE: Удаление пользователя")
    default:
        http.Error(w, "Метод не поддерживается", http.StatusMethodNotAllowed)
    }
}

Параметры из URL

func userHandler(w http.ResponseWriter, r *http.Request) {
    // Из пути: /user/123
    path := strings.TrimPrefix(r.URL.Path, "/user/")
    id, _ := strconv.Atoi(path)
    
    // GET-параметры: /user?id=123
    idQuery := r.URL.Query().Get("id")
    
    fmt.Fprintf(w, "ID из пути: %d, ID из query: %s", id, idQuery)
}

Middleware (промежуточные обработчики)

func loggingMiddleware(next http.HandlerFunc) http.HandlerFunc {
    return func(w http.ResponseWriter, r *http.Request) {
        start := time.Now()
        log.Printf("Запрос: %s %s", r.Method, r.URL.Path)
        next(w, r)
        log.Printf("Завершен за %v", time.Since(start))
    }
}

func main() {
    handler := loggingMiddleware(helloHandler)
    http.HandleFunc("/", handler)
    http.ListenAndServe(":8080", nil)
}

📦 Работа с JSON

Теги структур

type User struct {
    ID        int    `json:"id"`
    Name      string `json:"name"`
    Email     string `json:"email"`
    Password  string `json:"-"`                    // игнорировать
    CreatedAt string `json:"created_at,omitempty"` // omitempty — пропустить, если пусто
}

Маршалинг (структура → JSON)

func getUsersHandler(w http.ResponseWriter, r *http.Request) {
    users := []User{
        {ID: 1, Name: "Анна", Email: "anna@example.com"},
        {ID: 2, Name: "Борис", Email: "boris@example.com"},
    }
    
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(users)
}

Анмаршалинг (JSON → структура)

func createUserHandler(w http.ResponseWriter, r *http.Request) {
    var user User
    
    err := json.NewDecoder(r.Body).Decode(&user)
    if err != nil {
        http.Error(w, "Неверный JSON", http.StatusBadRequest)
        return
    }
    
    // Обработка пользователя...
    w.WriteHeader(http.StatusCreated)
    json.NewEncoder(w).Encode(user)
}

Удобные функции для JSON-ответов

func respondJSON(w http.ResponseWriter, status int, data interface{}) {
    w.Header().Set("Content-Type", "application/json")
    w.WriteHeader(status)
    json.NewEncoder(w).Encode(data)
}

func respondError(w http.ResponseWriter, status int, message string) {
    respondJSON(w, status, map[string]string{"error": message})
}

🗄️ Работа с базами данных

Подключение к PostgreSQL через database/sql

import (
    "database/sql"
    _ "github.com/lib/pq"
)

func main() {
    connStr := "user=postgres dbname=mydb password=123 sslmode=disable"
    db, err := sql.Open("postgres", connStr)
    if err != nil {
        log.Fatal(err)
    }
    defer db.Close()
    
    // Проверка подключения
    if err := db.Ping(); err != nil {
        log.Fatal(err)
    }
}

Выполнение запроса

type User struct {
    ID   int
    Name string
    Age  int
}

func getUsers(db *sql.DB) ([]User, error) {
    rows, err := db.Query("SELECT id, name, age FROM users")
    if err != nil {
        return nil, err
    }
    defer rows.Close()
    
    var users []User
    for rows.Next() {
        var u User
        if err := rows.Scan(&u.ID, &u.Name, &u.Age); err != nil {
            return nil, err
        }
        users = append(users, u)
    }
    return users, nil
}

Краткий обзор GORM (ORM)

import "gorm.io/gorm"

type Product struct {
    gorm.Model
    Code  string
    Price uint
}

func main() {
    db, _ := gorm.Open(sqlite.Open("test.db"), &gorm.Config{})
    
    // Автомиграция
    db.AutoMigrate(&Product{})
    
    // Создание
    db.Create(&Product{Code: "D42", Price: 100})
    
    // Чтение
    var product Product
    db.First(&product, 1)
    db.First(&product, "code = ?", "D42")
    
    // Обновление
    db.Model(&product).Update("Price", 200)
    
    // Удаление
    db.Delete(&product)
}
