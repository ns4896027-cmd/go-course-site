Лекция 6. Методы и интерфейсы

Длительность лекции: 60 минут
Уровень: Продвинутый
Темы: Методы, интерфейсы, пустой интерфейс, встраивание интерфейсов

📌 Методы в Go

Вместо классов, как в классических ООП-языках, в Go используются методы. Метод — это функция, связанная с определённым типом (обычно со структурой).

Сигнатура метода

func (название_параметра тип_получателя) название_метода(параметры) (типы_возвращаемых_значений) {
    // тело метода
}

Пример метода для структуры

type Person struct {
    Name string
    Age  int
}

// Метод с получателем-значением
func (p Person) Greet() string {
    return "Привет, меня зовут " + p.Name
}

// Метод с получателем-указателем (может изменять структуру)
func (p *Person) Birthday() {
    p.Age++
}

func main() {
    person := Person{Name: "Анна", Age: 25}
    fmt.Println(person.Greet())  // Привет, меня зовут Анна
    person.Birthday()
    fmt.Println(person.Age)  // 26
}

Методы для пользовательских типов

В Go можно создавать методы не только для структур, но и для любых пользовательских типов.

// Новый тип на основе int
type MyInt int

func (m MyInt) IsPositive() bool {
    return m > 0
}

func (m MyInt) Double() MyInt {
    return m * 2
}

// Новый тип на основе среза строк
type StringSlice []string

func (ss StringSlice) Join(separator string) string {
    result := ""
    for i, s := range ss {
        if i > 0 {
            result += separator
        }
        result += s
    }
    return result
}

func main() {
    var num MyInt = 42
    fmt.Println(num.IsPositive())  // true
    fmt.Println(num.Double())      // 84

    fruits := StringSlice{"яблоко", "груша", "банан"}
    fmt.Println(fruits.Join(", "))  // яблоко, груша, банан
}
Важное ограничение: Нельзя определять методы для встроенных типов напрямую. Только через создание нового типа.

Получатель-значение vs Получатель-указатель

ХарактеристикаПолучатель-значение (p Person)Получатель-указатель (p *Person)
Изменение полей❌ Не изменяет оригинал✅ Изменяет оригинал
КопированиеКопирует всю структуруКопирует только указатель (быстрее)
Вызов для nilБезопасноНужна проверка на nil
Правило выбора: Используйте получатель-указатель, если метод изменяет структуру или структура большая. В остальных случаях можно использовать получатель-значение.

Правила видимости

  • Маленькая буква — метод доступен только внутри пакета
  • Большая буква — метод экспортируется (доступен из других пакетов)

🔌 Интерфейсы

Интерфейс — это тип, который определяет набор методов. Любой другой тип, который реализует все эти методы, автоматически удовлетворяет интерфейсу.

Ключевая особенность Go: Реализация интерфейса происходит неявно (duck typing). Не нужно писать implements или extends.

Пример реализации интерфейса

// Определяем интерфейс Speaker
type Speaker interface {
    Speak() string
}

// Структура Dog
type Dog struct {
    Name string
}

// Dog реализует Speak — значит Dog реализует интерфейс Speaker
func (d Dog) Speak() string {
    return "Гав! Меня зовут " + d.Name
}

// Структура Cat
type Cat struct {
    Name string
}

// Cat реализует Speak — значит Cat тоже реализует Speaker
func (c Cat) Speak() string {
    return "Мяу! Я " + c.Name
}

// Функция, которая работает с любым Speaker
func makeSound(s Speaker) {
    fmt.Println(s.Speak())
}

func main() {
    dog := Dog{Name: "Шарик"}
    cat := Cat{Name: "Мурка"}
    
    makeSound(dog)  // Гав! Меня зовут Шарик
    makeSound(cat)  // Мяу! Я Мурка
}

Интерфейсы как контракты

Интерфейсы позволяют писать гибкий и переиспользуемый код. Функция принимает интерфейс, а не конкретный тип.

Пример: система оплаты

// Интерфейс PaymentMethod
type PaymentMethod interface {
    Pay(amount float64) string
}

// Банковская карта
type CreditCard struct {
    Number string
    Holder string
}

func (c CreditCard) Pay(amount float64) string {
    return fmt.Sprintf("Оплачено %.2f руб. картой %s", amount, c.Number)
}

// Наличные
type Cash struct{}

func (c Cash) Pay(amount float64) string {
    return fmt.Sprintf("Оплачено %.2f руб. наличными", amount)
}

// PayPal
type PayPal struct {
    Email string
}

func (p PayPal) Pay(amount float64) string {
    return fmt.Sprintf("Оплачено %.2f руб. через PayPal (%s)", amount, p.Email)
}

// Универсальная функция оплаты
func processPayment(p PaymentMethod, amount float64) {
    fmt.Println(p.Pay(amount))
}

func main() {
    card := CreditCard{Number: "1234-5678", Holder: "Иван"}
    cash := Cash{}
    paypal := PayPal{Email: "ivan@example.com"}
    
    processPayment(card, 1000)
    processPayment(cash, 500)
    processPayment(paypal, 750)
}

📦 Пустой интерфейс interface{} (any)

Пустой интерфейс не определяет ни одного метода. Это означает, что любой тип удовлетворяет пустому интерфейсу.

var anything interface{}  // может хранить значение любого типа
anything = 42
anything = "hello"
anything = Dog{Name: "Шарик"}
anything = []int{1, 2, 3}

Утверждение типа (Type Assertion)

Позволяет извлечь конкретное значение из интерфейса.

func describe(i any) {
    // Безопасное утверждение типа
    if s, ok := i.(string); ok {
        fmt.Printf("Это строка: %q\n", s)
    } else if n, ok := i.(int); ok {
        fmt.Printf("Это число: %d\n", n)
    } else {
        fmt.Printf("Неизвестный тип: %T\n", i)
    }
}

Type Switch

func getType(i any) {
    switch v := i.(type) {
    case string:
        fmt.Printf("Строка длины %d\n", len(v))
    case int:
        fmt.Printf("Целое число: %d\n", v)
    case bool:
        fmt.Printf("Логическое значение: %t\n", v)
    default:
        fmt.Printf("Неизвестный тип: %T\n", v)
    }
}
Совет: В Go 1.18+ появился псевдоним any для interface{}. Используйте any в современном коде.

🔄 Встраивание интерфейсов

Go позволяет создавать композитные интерфейсы путём встраивания других интерфейсов. Это похоже на композицию в ООП.

// Базовые интерфейсы
type Reader interface {
    Read(p []byte) (n int, err error)
}

type Writer interface {
    Write(p []byte) (n int, err error)
}

type Closer interface {
    Close() error
}

// Встраивание интерфейсов
type ReadWriter interface {
    Reader   // встраиваем Reader
    Writer   // встраиваем Writer
}

type ReadWriteCloser interface {
    Reader
    Writer
    Closer
}

Пример с геометрическими фигурами

type Shape interface {
    Area() float64
}

type Perimeter interface {
    Perimeter() float64
}

type Geometry interface {
    Shape
    Perimeter
    Name() string
}

type Circle struct {
    Radius float64
}

func (c Circle) Area() float64 {
    return math.Pi * c.Radius * c.Radius
}

func (c Circle) Perimeter() float64 {
    return 2 * math.Pi * c.Radius
}

func (c Circle) Name() string {
    return "Круг"
}

func describe(g Geometry) {
    fmt.Printf("%s: площадь = %.2f, периметр = %.2f\n",
        g.Name(), g.Area(), g.Perimeter())
}
